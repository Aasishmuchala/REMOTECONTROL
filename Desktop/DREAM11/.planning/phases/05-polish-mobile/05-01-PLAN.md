**1. Upgrade GlassCard to support interactive mode (per D-09, Pitfall 3)**

Replace `coinxi/src/components/GlassCard.tsx` with:

```tsx
'use client'

import { motion } from 'framer-motion'
import { cn } from '@/lib/utils'

interface GlassCardProps {
  children: React.ReactNode
  className?: string
  elevated?: boolean
  interactive?: boolean
}

export default function GlassCard({
  children,
  className,
  elevated = false,
  interactive = false,
}: GlassCardProps) {
  if (interactive) {
    return (
      <motion.div
        whileTap={{ scale: 0.97 }}
        transition={{ type: 'spring', stiffness: 400, damping: 30 }}
        className={cn(
          'rounded-2xl p-4',
          elevated ? 'glass-elevated' : 'glass',
          'glow-hover cursor-pointer',
          className
        )}
      >
        {children}
      </motion.div>
    )
  }

  return (
    <div
      className={cn(
        'rounded-2xl p-4',
        elevated ? 'glass-elevated' : 'glass',
        className
      )}
    >
      {children}
    </div>
  )
}
```

Key points:
- `interactive` prop is opt-in (default false) — non-interactive cards remain plain divs
- Uses `motion.div` for interactive cards (Pitfall 3: whileTap requires motion element)
- Applies `glow-hover` CSS class (from Task 1) for hover glow
- Spring constants match established codebase pattern: stiffness 400, damping 30
- `'use client'` directive added since framer-motion is used
- Non-interactive path remains a plain `<div>` — no extra bundle for static cards
